import { createFlipbook } from './lib/folio3d.js';

const book = createFlipbook(element, {
  source: { type: 'pdf', data: './catalog.pdf' },
  theme: 'modern',
});

await book.ready;
book.goToPage(4);

// When the view unmounts
book.destroy();

  --folio-font: Georgia, serif;
}
